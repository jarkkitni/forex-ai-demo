ร้านที่ลูกค้าต่างชาติเยอะ เช่น ร้านอาหารย่านท่องเที่ยว สปา ร้านแลกเงิน คลินิกเสริมความงาม มักเจอปัญหาเดียวกัน คือคนยืนรอหน้าร้านโดยไม่รู้ว่าอีกนานแค่ไหน พนักงานต้องอธิบายเป็นภาษาอังกฤษซ้ำๆ และพอถึงสิ้นเดือนก็ไม่มีตัวเลขว่าวันไหนคนเยอะ บทความนี้เล่าโครงสร้างของระบบคิวแบบสแกน QR ที่ลูกค้าไม่ต้องโหลดแอป รองรับหลายภาษา และดึงรายงานย้อนหลังได้

ภาพรวมการทำงาน

  1. ร้านติด QR ไว้หน้าร้าน (QR เดียวใช้ได้ตลอด ไม่ต้องพิมพ์ใหม่ทุกวัน)
  2. ลูกค้าสแกน → เปิดหน้าเว็บในมือถือ → เลือกภาษา → เลือกบริการและจำนวนคน → ได้เลขคิว
  3. หน้ามือถือของลูกค้าแสดงเลขคิว จำนวนคิวก่อนหน้า และอัปเดตเองเมื่อใกล้ถึงคิว
  4. พนักงานมีหน้าจอเรียกคิว กด "เรียกคิวถัดไป" / "ข้าม" / "เสร็จแล้ว"
  5. ทุกการกดถูกบันทึกเวลาไว้ เอาไปทำรายงานได้

ทำไมเป็นหน้าเว็บ ไม่ใช่แอป

นักท่องเที่ยวไม่โหลดแอปเพื่อรอคิวครั้งเดียว และไม่ใช่ทุกคนใช้ LINE (ลูกค้าจีนใช้ WeChat ลูกค้ายุโรปใช้ WhatsApp) หน้าเว็บเปิดได้จากกล้องมือถือทุกรุ่นทันที นี่คือเหตุผลหลักที่ระบบคิวสำหรับลูกค้าต่างชาติควรเป็นเว็บ

เรื่องภาษา

  • เดาภาษาเริ่มต้นจากภาษาของเบราว์เซอร์ (navigator.language) แล้วมีปุ่มเปลี่ยนภาษาไว้มุมจอเสมอ
  • ข้อความในระบบคิวมีไม่กี่สิบประโยค แปลเก็บเป็นไฟล์ตามภาษา อย่าใช้แปลอัตโนมัติสดๆ เพราะคำแปลเพี้ยนและช้า
  • ชื่อบริการให้ร้านกรอกได้หลายภาษาในหน้าหลังบ้าน
  • ตัวเลขคิวใช้เลขอารบิก (1, 2, 3) ทุกภาษา อ่านง่ายที่สุด

ข้อมูลที่ต้องเก็บ

create table queue_tickets (
  id           bigserial primary key,
  branch_id    int not null,
  ticket_no    int not null,           -- เลขคิวของวันนั้น เริ่ม 1 ใหม่ทุกวัน
  service      text not null,
  party_size   int default 1,
  lang         text,                   -- ภาษาที่ลูกค้าเลือก ใช้ดูสัดส่วนลูกค้าต่างชาติ
  status       text not null default 'waiting',  -- waiting / called / done / skipped
  created_at   timestamptz default now(),
  called_at    timestamptz,
  done_at      timestamptz,
  queue_date   date not null default (now() at time zone 'Asia/Bangkok')::date,
  unique (branch_id, queue_date, ticket_no)
);

สองเรื่องที่ต้องระวัง:

  • เลขคิวซ้ำ — ถ้าลูกค้าสองคนสแกนพร้อมกัน อาจได้เลขเดียวกัน ให้ unique ในฐานข้อมูลกันไว้ แล้วออกเลขใหม่เมื่อชน
  • วันที่ต้องเป็นเวลาไทย — ถ้าเซิร์ฟเวอร์ใช้ UTC เลขคิวจะเริ่มนับ 1 ใหม่ตอน 7 โมงเช้า ไม่ใช่เที่ยงคืน

ระบบคิวไม่จำเป็นต้องเก็บชื่อหรือเบอร์โทรลูกค้า ถ้าไม่เก็บก็ไม่ต้องกังวลเรื่องข้อมูลส่วนบุคคลมากนัก ถ้าจำเป็นต้องเก็บ อ่าน PDPA สำหรับร้านออนไลน์

รายงานที่เจ้าของร้านอยากเห็นจริงๆ

  • จำนวนคิวต่อวัน / ต่อชั่วโมง — ใช้จัดตารางพนักงาน
  • เวลารอเฉลี่ย = called_at - created_at
  • เวลาให้บริการเฉลี่ย = done_at - called_at แยกตามบริการ
  • อัตราลูกค้าหนี = คิวที่ถูกข้าม ÷ คิวทั้งหมด ถ้าสูงตอนเวลาไหน แปลว่าช่วงนั้นรอนานเกินไป
  • สัดส่วนภาษา — บอกว่าลูกค้าชาติไหนมาเยอะ ใช้วางแผนโฆษณาหรือจ้างพนักงานที่พูดภาษานั้นได้
-- เวลารอเฉลี่ยรายชั่วโมง ของ 30 วันล่าสุด
select extract(hour from created_at at time zone 'Asia/Bangkok') as hr,
       count(*) as tickets,
       round(avg(extract(epoch from called_at - created_at)) / 60) as avg_wait_min
from queue_tickets
where queue_date >= current_date - 30 and called_at is not null
group by hr order by hr;

ให้ปุ่ม export เป็น Excel/CSV ด้วย เจ้าของร้านหลายคนชอบเปิดดูเองมากกว่าดูกราฟในระบบ

อัปเดตหน้าลูกค้าแบบไม่เปลืองเซิร์ฟเวอร์

ไม่จำเป็นต้องใช้ WebSocket ร้านทั่วไปให้หน้ามือถือถามสถานะทุก 10–15 วินาทีก็พอ และพอเหลืออีก 2 คิว ให้หน้าจอสั่นหรือเปลี่ยนสีเตือน ลูกค้าจะได้เดินกลับมาทัน
